cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(card_data_recovery C CXX ASM)
set(CMAKE_C_STANDARD 11)
option(RECOVERY_EXECUTE_ERASE "Enable the destructive data-partition erase" OFF)
pico_sdk_init()
add_executable(card_data_recovery main.c guard.c)
target_link_libraries(card_data_recovery pico_stdlib pico_bootrom hardware_irq)
if(RECOVERY_EXECUTE_ERASE)
    target_compile_definitions(card_data_recovery PRIVATE RECOVERY_EXECUTE_ERASE=1)
else()
    target_compile_definitions(card_data_recovery PRIVATE RECOVERY_EXECUTE_ERASE=0)
endif()
pico_set_binary_type(card_data_recovery no_flash)
pico_add_extra_outputs(card_data_recovery)
