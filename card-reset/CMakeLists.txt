cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(card_data_recovery C CXX ASM)
set(CMAKE_C_STANDARD 11)
option(RECOVERY_EXECUTE_ERASE "Enable the destructive data-partition erase" OFF)
pico_sdk_init()
add_executable(card_data_recovery main.c guard.c)
set_source_files_properties(main.c guard.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-fstack-usage")
target_link_libraries(card_data_recovery pico_stdlib pico_bootrom hardware_irq)
target_compile_definitions(card_data_recovery PRIVATE PICO_STACK_SIZE=0x1000)
if(RECOVERY_EXECUTE_ERASE)
    target_compile_definitions(card_data_recovery PRIVATE RECOVERY_EXECUTE_ERASE=1)
else()
    target_compile_definitions(card_data_recovery PRIVATE RECOVERY_EXECUTE_ERASE=0)
endif()
pico_set_binary_type(card_data_recovery no_flash)
pico_add_extra_outputs(card_data_recovery)
foreach(stage 0 1)
    add_executable(stage_probe_${stage} stage_probe.c)
    target_link_libraries(stage_probe_${stage} pico_stdlib pico_bootrom)
    target_compile_definitions(stage_probe_${stage} PRIVATE PROBE_CHECK_PARTITION=${stage} PICO_STACK_SIZE=0x1000)
    pico_set_binary_type(stage_probe_${stage} no_flash)
    pico_add_extra_outputs(stage_probe_${stage})
endforeach()
